\section{Deployment Script}

\begin{lstlisting}[language=Solidity, caption={Deploy.s.sol - Contract Deployment Script}]
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import "forge-std/Script.sol";
import "../src/Profile.sol";
import "../src/Jobs.sol";
import "../src/Proposals.sol";

contract DeployScript is Script {
    function run() external {
        vm.startBroadcast();
        
        // Deploy UserProfile contract
        UserProfile profile = new UserProfile();
        
        // Deploy JobsContract
        JobsContract jobs = new JobsContract();
        
        // Deploy ProposalsContract
        ProposalsContract proposals = new ProposalsContract();
        
        // Link contracts
        jobs.setProposalsContractAddress(address(proposals));
        proposals.setJobsContractAddress(address(jobs));
        
        vm.stopBroadcast();
        
        // Log deployed addresses
        console.log("UserProfile deployed at:", address(profile));
        console.log("JobsContract deployed at:", address(jobs));
        console.log("ProposalsContract deployed at:", address(proposals));
    }
}
\end{lstlisting}

\section{Pinata IPFS Setup}

\subsection{Getting Pinata JWT}

\begin{enumerate}
    \item Create account at \url{https://www.pinata.cloud/}
    \item Navigate to API Keys section
    \item Create new API key with permissions:
    \begin{itemize}
        \item pinFileToIPFS
        \item pinJSONToIPFS
        \item unpin
    \end{itemize}
    \item Copy the JWT token for frontend configuration
\end{enumerate}

\subsection{Pinata Gateway Configuration}

\begin{enumerate}
    \item In Pinata dashboard, go to Gateway section
    \item Create a dedicated gateway (paid plan) or use public gateway
    \item Configure gateway URL in frontend environment variables
    \item Format: \texttt{gateway.pinata.cloud} or your custom domain
\end{enumerate}
